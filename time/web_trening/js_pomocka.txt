/*Таймеры и интервалы*/

setInterval(первый параметр - имя фукнции, второй параметр - количество милисекунд); - функция которая создаёт какой то интервал

clearInterval(название переменной); - для очистки и остановки интервала

// для того что бы остановить счётчик интервала нужно его поместить в какую то переменную и в будущем, когда нужно будет остановить его, то вызвать функцию clearInterval() с переданным названием переменной.

setTimeout(первый параметр - имя фукнции(можно анонимную(безимянную)), второй параметр - количество милисекунд); - для создания таймера, выполняется один раз по достижении установленного времени и больше не работает.
